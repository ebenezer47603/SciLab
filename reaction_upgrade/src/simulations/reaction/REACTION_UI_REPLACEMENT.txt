3) Replace initializeReactionSimulator() with:

function initializeReactionSimulator() {

    const container =
        document.getElementById(
            "reaction-container"
        );

    if (!container) {
        console.warn(
            "Reaction: #reaction-container not found."
        );
        return;
    }

    let simulator = null;
    let renderer = null;
    let animationFrame = null;
    let resizeHandler = null;

    try {

        simulator =
            new ReactionSimulator();

        const scene =
            simulator.getScene();

        const camera =
            simulator.getCamera();

        renderer =
            new THREE.WebGLRenderer({
                antialias: true,
                alpha: false
            });

        renderer.setPixelRatio(
            Math.min(
                window.devicePixelRatio || 1,
                2
            )
        );

        renderer.outputColorSpace =
            THREE.SRGBColorSpace;

        renderer.shadowMap.enabled =
            true;

        container.innerHTML = "";

        container.appendChild(
            renderer.domElement
        );

        const resize = () => {

            const width =
                Math.max(
                    320,
                    container.clientWidth
                );

            const height =
                Math.max(
                    420,
                    container.clientHeight
                );

            camera.aspect =
                width / height;

            camera.updateProjectionMatrix();

            renderer.setSize(
                width,
                height
            );
        };

        resizeHandler = resize;

        window.addEventListener(
            "resize",
            resize
        );

        resize();

        const select =
            document.getElementById(
                "reaction-select"
            );

        const temperatureInput =
            document.getElementById(
                "reaction-temperature"
            );

        const temperatureValue =
            document.getElementById(
                "reaction-temperature-value"
            );

        const rateValue =
            document.getElementById(
                "reaction-rate-value"
            );

        const progressValue =
            document.getElementById(
                "reaction-progress-value"
            );

        const progressBar =
            document.getElementById(
                "reaction-progress-bar"
            );

        const equation =
            document.getElementById(
                "reaction-equation"
            );

        const selectedName =
            document.getElementById(
                "reaction-selected-name"
            );

        const moleculeCount =
            document.getElementById(
                "reaction-molecule-count"
            );

        const updateReactionInfo = () => {

            const reaction =
                simulator.getReaction();

            if (selectedName) {
                selectedName.textContent =
                    reaction.name;
            }

            if (equation) {
                equation.textContent =
                    reaction.equation;
            }
        };

        const updateTemperature = () => {

            const value =
                Number(
                    temperatureInput?.value || 298
                );

            if (temperatureValue) {
                temperatureValue.textContent =
                    `${value} K`;
            }

            simulator.setTemperature(
                value
            );
        };

        const updateUI = () => {

            const engine =
                simulator.getEngine();

            const counts =
                simulator
                    .getMolecules()
                    .getCounts();

            const reaction =
                simulator.getReaction();

            const lines = [];

            for (
                const item
                of reaction.reactants
            ) {
                const info =
                    getMolecule(item.formula);

                lines.push(
                    `${info?.formula || item.formula}: ${counts[item.formula] || 0}`
                );
            }

            for (
                const item
                of reaction.products
            ) {
                const info =
                    getMolecule(item.formula);

                lines.push(
                    `${info?.formula || item.formula}: ${counts[item.formula] || 0}`
                );
            }

            if (moleculeCount) {
                moleculeCount.innerHTML =
                    lines.join("<br>");
            }

            if (rateValue) {
                rateValue.textContent =
                    `${engine.getReactionRatePercent()}%`;
            }

            if (progressValue) {
                progressValue.textContent =
                    `${engine.getProgressPercent()}%`;
            }

            if (progressBar) {
                progressBar.style.width =
                    `${engine.getProgressPercent()}%`;
            }
        };

        if (select) {

            select.addEventListener(
                "change",
                () => {

                    simulator.setReaction(
                        select.value
                    );

                    updateReactionInfo();
                    updateTemperature();
                    updateUI();
                }
            );
        }

        document
            .getElementById(
                "reaction-start"
            )
            ?.addEventListener(
                "click",
                () => simulator.start()
            );

        document
            .getElementById(
                "reaction-pause"
            )
            ?.addEventListener(
                "click",
                () => simulator.pause()
            );

        document
            .getElementById(
                "reaction-reset"
            )
            ?.addEventListener(
                "click",
                () => {
                    simulator.reset();
                    updateTemperature();
                    updateReactionInfo();
                    updateUI();
                }
            );

        temperatureInput?.addEventListener(
            "input",
            () => {
                updateTemperature();
                updateUI();
            }
        );

        updateReactionInfo();
        updateTemperature();
        updateUI();

        let lastTime =
            performance.now();

        const animate =
            currentTime => {

                if (
                    !simulator ||
                    !renderer
                ) {
                    return;
                }

                const deltaTime =
                    Math.min(
                        0.05,
                        Math.max(
                            0,
                            (
                                currentTime -
                                lastTime
                            ) / 1000
                        )
                    );

                lastTime =
                    currentTime;

                simulator.update(
                    deltaTime
                );

                updateUI();

                renderer.render(
                    scene,
                    camera
                );

                animationFrame =
                    requestAnimationFrame(
                        animate
                    );
            };

        animationFrame =
            requestAnimationFrame(
                animate
            );

        currentCleanup = () => {

            if (
                animationFrame !== null
            ) {
                cancelAnimationFrame(
                    animationFrame
                );

                animationFrame = null;
            }

            if (resizeHandler) {
                window.removeEventListener(
                    "resize",
                    resizeHandler
                );

                resizeHandler = null;
            }

            if (renderer) {
                renderer.dispose();

                renderer.domElement
                    ?.remove();

                renderer = null;
            }

            simulator = null;
        };

        console.log(
            "Reaction Laboratory initialized."
        );

    } catch (error) {

        console.error(
            "Reaction Laboratory error:",
            error
        );

        if (renderer) {
            renderer.dispose();
        }

        renderer = null;
        simulator = null;
    }
}

============================================================
IMPORTANT FIX
============================================================

The existing MoleculeRenderer only knew H2, O2 and H2O.
The replacement above uses the molecule database and therefore supports
all 22 molecular/element definitions needed by the 10 reactions.

The existing ReactionEngine also hard-coded H2 + O2 and manually created
H2O. The replacement engine uses reaction.reactants/products and
stoichiometric coefficients instead.

Do NOT import a second ReactionControls.js into main.js.
